function ensureDir(p) {
  fs.mkdirSync(p, { recursive: true });

function listPngs(dir, out) {
  if (!fs.existsSync(dir)) return;
  const entries = fs.readdirSync(dir, { withFileTypes: true });
  for (const ent of entries) {
    const full = path.join(dir, ent.name);
    if (ent.isDirectory()) {
      if (ent.name.toLowerCase() === "_atlas") continue;
      listPngs(full, out);
      continue;
    }
    if (!ent.isFile()) continue;
    if (!ent.name.toLowerCase().endsWith(".png")) continue;
    out.push(full);
  }
}

function basenameNoExt(p) {
  const file = p.split(/[\\/]/).pop() || p;
  return file.replace(/\.png$/i, "");
}

function parseWeaponFilename(base) {
  const parts = base.split("__").filter(Boolean);
  if (parts.length < 6) return null;
  const tPart = parts[0];
  if (!TILE_RE.test(tPart)) return null;
  const category = parts[1];
  let vIndex = -1;
  for (let i = parts.length - 1; i >= 0; i--) {
    if (/^v.+/i.test(parts[i])) {
      vIndex = i;
      break;
    }
  }
  if (vIndex < 5) return null;
  const layer = parts[vIndex - 1];
  const anim = parts[vIndex - 2];
  const modelTokens = parts.slice(2, vIndex - 2);
  if (!modelTokens.length) return null;
  const model = modelTokens.join("_");
  const variant = parts[vIndex].slice(1);
  const tileNum = Number(tPart.slice(1));
  const tile = (tileNum === 64 ? 64 : tileNum === 128 ? 128 : 192);
  if (!category || !model || !anim || !variant) return null;
  if (layer !== "bg" && layer !== "fg") return null;
  return {
    key: base,
    tile,
    category,
    model,
    anim,
    layer,
    variant
  };
}

function sortEntries(a, b) {
  if (a.variant !== b.variant) return a.variant.localeCompare(b.variant);
  if (a.layer !== b.layer) return a.layer.localeCompare(b.layer);
  return a.key.localeCompare(b.key);
}

function copyPngIntoAtlas(src, atlas, dstX, dstY) {
  const w = src.width | 0;
  const h = src.height | 0;
  for (let y = 0; y < h; y++) {
    const srcRow = (y * w) << 2;
    const dstRow = ((dstY + y) * atlas.width + dstX) << 2;
    for (let x = 0; x < w; x++) {
      const si = srcRow + (x << 2);
      const di = dstRow + (x << 2);
      atlas.data[di] = src.data[si];
      atlas.data[di + 1] = src.data[si + 1];
      atlas.data[di + 2] = src.data[si + 2];
      atlas.data[di + 3] = src.data[si + 3];
    }
  }
}

// Optional auto-anchor: align the weapon handle (side opposite attack direction)
// to the hero's visual center. Cached per layer+frame+dir to avoid repeats.
const WEAPON_HANDLE_ADJUST_BY_DIR: Record<Dir4, { x: number; y: number }> = {
  up: { x: 0, y: 0 },
  down: { x: 0, y: 0 },
  left: { x: 0, y: 0 },
  right: { x: 0, y: 0 }
};

const _WEAPON_ANCHOR_CACHE = new Map<string, { dx: number; dy: number; hx: number; hy: number }>();
const _WEAPON_ANCHOR_LOG_ONCE = new Set<string>();

function _weaponAnchorKey(layerKey: string, frameIndex: number, dir: Dir4): string {
  return `${layerKey}::${frameIndex}::${dir}`;
}

function _scanHandlePoint(
  data: Uint8ClampedArray,
  w: number,
  h: number,
  dir: Dir4
): { hx: number; hy: number } {
  let sumX = 0;
  let sumY = 0;
  let count = 0;

  if (dir === "right") {
    // handle on left edge
    for (let y = 0; y < h; y++) {
      const row = y * w;
      for (let x = 0; x < w; x++) {
        const a = data[((row + x) << 2) + 3];
        if (a) {
          sumX += x;
          sumY += y;
          count++;
          break;
        }
      }
    }
  } else if (dir === "left") {
    // handle on right edge
    for (let y = 0; y < h; y++) {
      const row = y * w;
      for (let x = w - 1; x >= 0; x--) {
        const a = data[((row + x) << 2) + 3];
        if (a) {
          sumX += x;
          sumY += y;
          count++;
          break;
        }
      }
    }
  } else if (dir === "down") {
    // handle on top edge
    for (let x = 0; x < w; x++) {
      for (let y = 0; y < h; y++) {
        const a = data[((y * w + x) << 2) + 3];
        if (a) {
          sumX += x;
          sumY += y;
          count++;
          break;
        }
      }
    }
  } else {
    // dir === "up": handle on bottom edge
    for (let x = 0; x < w; x++) {
      for (let y = h - 1; y >= 0; y--) {
        const a = data[((y * w + x) << 2) + 3];
        if (a) {
          sumX += x;
          sumY += y;
          count++;
          break;
        }
      }
    }
  }

  if (!count) {
    return { hx: (w - 1) * 0.5, hy: (h - 1) * 0.5 };
  }

  return { hx: sumX / count, hy: sumY / count };
}

function _getWeaponHandleOffset(
  scene: Phaser.Scene,
  layerRef: WeaponSheetRef,
  frameIndex: number,
  dir: Dir4,
  dbgOn: boolean
): { dx: number; dy: number } {
  const key = _weaponAnchorKey(layerRef.key, frameIndex, dir);
  const cached = _WEAPON_ANCHOR_CACHE.get(key);
  if (cached) return { dx: cached.dx, dy: cached.dy };

  const textures: any = scene?.textures;
  const tex: any = textures?.get?.(layerRef.key);
  const fr: any = tex?.get?.(frameIndex);
  if (!tex || !fr) {
    const fallback = { dx: 0, dy: 0, hx: 0, hy: 0 };
    _WEAPON_ANCHOR_CACHE.set(key, fallback);
    return { dx: 0, dy: 0 };
  }

  const fw = (fr.cutWidth ?? fr.width ?? 0) | 0;
  const fh = (fr.cutHeight ?? fr.height ?? 0) | 0;
  const fx = (fr.cutX ?? fr.x ?? 0) | 0;
  const fy = (fr.cutY ?? fr.y ?? 0) | 0;

  const srcImg: any = fr?.texture?.source?.[0]?.image ?? tex?.source?.[0]?.image;
  if (!srcImg || fw <= 0 || fh <= 0) {
    const fallback = { dx: 0, dy: 0, hx: 0, hy: 0 };
    _WEAPON_ANCHOR_CACHE.set(key, fallback);
    return { dx: 0, dy: 0 };
  }

  const canvas = document.createElement("canvas");
  canvas.width = fw;
  canvas.height = fh;
  const g = canvas.getContext("2d", { willReadFrequently: true } as any);
  if (!g) {
    const fallback = { dx: 0, dy: 0, hx: 0, hy: 0 };
    _WEAPON_ANCHOR_CACHE.set(key, fallback);
    return { dx: 0, dy: 0 };
  }

  g.clearRect(0, 0, fw, fh);
  g.drawImage(srcImg, fx, fy, fw, fh, 0, 0, fw, fh);
  const imgData = g.getImageData(0, 0, fw, fh);
  const { hx, hy } = _scanHandlePoint(imgData.data, fw, fh, dir);

  const cx = (fw - 1) * 0.5;
  const cy = (fh - 1) * 0.5;
  const adj = WEAPON_HANDLE_ADJUST_BY_DIR[dir] ?? { x: 0, y: 0 };
  const dx = Math.round(cx - hx + adj.x);
  const dy = Math.round(cy - hy + adj.y);

  _WEAPON_ANCHOR_CACHE.set(key, { dx, dy, hx, hy });

  if (dbgOn && !_WEAPON_ANCHOR_LOG_ONCE.has(key)) {
    _WEAPON_ANCHOR_LOG_ONCE.add(key);
    console.log("[WPN-ANCHOR]", {
      key,
      frameIndex,
      dir,
      fw,
      fh,
      hx: Math.round(hx),
      hy: Math.round(hy),
      dx,
      dy
    });
  }

  return { dx, dy };
}

function _legacyWeaponPathForSheetKey(sheetKey: string): { url: string; key: string } | null {
    const cleanKey = String(sheetKey || "").replace(/__dup\\d+$/i, "");
    const parts = cleanKey.split("__").filter(Boolean);
    if (parts.length < 6) return null;
    const tileTag = parts[0];
    const category = parts[1];
    if (!tileTag || !category) return null;
    const url = `assets/weapons/${tileTag}/${category}/${cleanKey}.png`;
    return { url, key: cleanKey };
}

function _ensureLegacyWeaponSheet(
    sc: any,
    sheetKey: string,
    frameW: number,
    frameH: number
): string | null {
    const info = _legacyWeaponPathForSheetKey(sheetKey);
    if (!info) return null;
    const legacyKey = `legacy__${info.key}`;
    const textures: any = sc?.textures;
    if (textures && typeof textures.exists === "function" && textures.exists(legacyKey)) return legacyKey;
    const queued = queueSpritesheetOnce(sc, legacyKey, info.url, frameW, frameH);
    if (queued) {
        try {
            const loader: any = sc?.load;
            if (loader && typeof loader.start === "function") loader.start();
        } catch { /* ignore */ }
    }
    return legacyKey;
}

function _dbgLayerSnapshot(
    sc: any,
    nativeHero: Phaser.GameObjects.Sprite,
    label: string,
    centerX: number,
    centerY: number,
    radiusPx: number
): void {
    if (!DEBUG_LAYER_SNAPSHOT) return;
    if (!sc || !nativeHero) return;
    const list: any[] = (sc.children && typeof sc.children.getChildren === "function")
        ? (sc.children.getChildren() as any[])
        : ((sc.children && sc.children.list) ? sc.children.list : []);
    if (!list || !list.length) return;

    const rect = {
        x: (centerX | 0) - (radiusPx | 0),
        y: (centerY | 0) - (radiusPx | 0),
        w: (radiusPx | 0) * 2,
        h: (radiusPx | 0) * 2
    };

    const items: any[] = [];
    for (let i = 0; i < list.length; i++) {
        const obj: any = list[i];
        if (!obj) continue;
        if (typeof obj.getBounds !== "function") continue;
        const b = obj.getBounds();
        if (!b) continue;
        const intersects =
            b.x < rect.x + rect.w &&
            b.x + b.width > rect.x &&
            b.y < rect.y + rect.h &&
            b.y + b.height > rect.y;
        if (!intersects) continue;
        const texKey = String(obj.texture?.key ?? "");
        const frame = obj.frame?.name ?? obj.frame?.index ?? 0;
        items.push({
            i,
            depth: (obj.depth ?? 0),
            vis: !!obj.visible,
            alpha: typeof obj.alpha === "number" ? Number(obj.alpha.toFixed(2)) : 1,
            tex: texKey,
            frame,
            x: Math.round(obj.x ?? 0),
            y: Math.round(obj.y ?? 0),
            name: String(obj.name ?? "")
        });
    }

    items.sort((a, b) => (a.depth - b.depth) || (a.i - b.i));
    const trimmed = items.slice(0, Math.max(1, LAYER_SNAPSHOT_MAX_ITEMS));

    console.log("[LAYER][SNAP]", {
        label,
        center: { x: Math.round(centerX), y: Math.round(centerY) },
        radius: radiusPx | 0,
        count: items.length,
        items: trimmed
    });
}

function _getRowNonEmptyFlags(scene: Phaser.Scene, ref: WeaponSheetRef, rows: number, cols: number): boolean[] | null {
  if (!ref?.key) return null;
  const cached = ROW_NONEMPTY_CACHE.get(ref.key);
  if (cached) return cached;
  try {
    const tex = scene.textures.get(ref.key);
    const img: any = tex?.getSourceImage?.();
    if (!img || !img.width || !img.height) return null;
    const tileW = ref.frameW | 0;
    const tileH = ref.frameH | 0;
    if (tileW <= 0 || tileH <= 0) return null;
    const w = img.width | 0;
    const h = img.height | 0;
    const expectedW = (cols | 0) * (tileW | 0);
    const expectedH = (rows | 0) * (tileH | 0);
    if (expectedW <= 0 || expectedH <= 0) return null;

    const canvas = document.createElement("canvas");
    canvas.width = w;
    canvas.height = h;
    const ctx = canvas.getContext("2d");
    if (!ctx) return null;
    ctx.drawImage(img, 0, 0);
    const data = ctx.getImageData(0, 0, w, h).data;

    const flags: boolean[] = new Array(rows).fill(false);
    for (let row = 0; row < rows; row++) {
      const y0 = row * tileH;
      const y1 = Math.min(h, y0 + tileH);
      let found = false;
      for (let y = y0; y < y1 && !found; y++) {
        const rowOff = (y * w) << 2;
        for (let x = 0; x < expectedW && x < w; x++) {
          const a = data[rowOff + (x << 2) + 3];
          if (a) { found = true; break; }
        }
      }
      flags[row] = found;
    }
    ROW_NONEMPTY_CACHE.set(ref.key, flags);
    return flags;
  } catch {
    return null;
  }
}

function _weaponRowForDir(scene: Phaser.Scene, ref: WeaponSheetRef, dir: Dir4, rows: number, cols: number): number {
  const baseRow = rows >= 4 ? dirIndex(dir) : 0;
  if (baseRow <= 0) return baseRow;
  if (String(ref?.layer || "") !== "bg") return baseRow;
  const flags = _getRowNonEmptyFlags(scene, ref, rows, cols);
  if (!flags || !flags.length) return baseRow;
  if (flags[baseRow]) return baseRow;
  if (flags[0]) return 0;
  return baseRow;
}

Audit of genweaponmodelatlas.mjs and why the grid isn’t the real culprit:

What the script actually does (and doesn’t do)

It never changes the internal frame order of a weapon sheet. Each source PNG is copied as‑is into the atlas, and the atlas JSON frame for that sheet uses the source PNG’s exact w/h.
The “grid” is only the packing layout for those whole sheets. It does not affect the row/col order inside each sheet.
At runtime, textures.addSpriteSheetFromAtlas(...) uses the atlas frame that points to the original sheet region, so the internal frame layout should match the original file exactly.
So if you’re seeing misalignment only in strength prep frames (0–2), the grid packing itself is almost certainly not the cause.

Actual atlas‑script risks that can cause real issues

Non‑divisible sheet sizes: If any sheet is not divisible by its tile size, the script floors cols/rows and effectively drops columns/rows. It only logs a warning; it doesn’t fix it. That will break frame selection. (See warnings.push("[weapon-atlas] not divisible...") in genweaponmodelatlas.mjs.)
Inconsistent cols/rows across variants/layers: The generator does not enforce identical cols/rows for all variants of the same model+anim. If one variant has fewer columns than another, the frame mapping will diverge by variant even if filenames match.
Bad filename parsing: If parseWeaponFilename misinterprets a model/anim/layer token, it will group the sheet under the wrong atlasKey, and you’ll resolve the wrong sheet later.

You can absolutely fix the column count, but you cannot make the atlas arbitrarily long because of GPU max texture size. In WebGL that’s commonly 4096 or 8192 (sometimes 16384 on good hardware), and exceeding it will fail texture upload.
